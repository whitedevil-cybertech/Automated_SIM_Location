def _get_nested(document: Dict[str, Any], path: str) -> Any:
    current: Any = document
    for part in path.split("."):
        if not isinstance(current, dict) or part not in current:
            return None
        current = current[part]
    return current


def _set_nested(document: Dict[str, Any], path: str, value: Any) -> None:
    current = document
    parts = path.split(".")
    for part in parts[:-1]:
        if part not in current or not isinstance(current[part], dict):
            current[part] = {}
        current = current[part]
    current[parts[-1]] = value


def _matches(document: Dict[str, Any], query: Dict[str, Any]) -> bool:
    for key, expected in query.items():
        if _get_nested(document, key) != expected:
            return False
    return True

class FakeCursor:
    def __init__(self, documents: list[Dict[str, Any]]) -> None:
        self.documents = documents

    def sort(self, key: str, direction: int):
        self.documents.sort(
            key=lambda document: document.get(key) or "",
            reverse=direction < 0,
        )
        return self

    def __aiter__(self):
        async def iterate():
            for document in self.documents:
                yield copy.deepcopy(document)

        return iterate()

class FakeCollection:
    def __init__(self) -> None:
        self.documents: list[Dict[str, Any]] = []

    async def insert_one(self, document: Dict[str, Any]):
        self.documents.append(copy.deepcopy(document))
        return SimpleNamespace(inserted_id=len(self.documents))

    async def find_one(
        self,
        query: Dict[str, Any],
        projection: Dict[str, int] | None = None,
    ):
        for document in self.documents:
            if _matches(document, query):
                return copy.deepcopy(document)
        return None

    def find(self, query: Dict[str, Any]) -> FakeCursor:
        matching_documents = [
            copy.deepcopy(document)
            for document in self.documents
            if _matches(document, query)
        ]
        return FakeCursor(matching_documents)

    async def update_one(
        self,
        query: Dict[str, Any],
        update: Dict[str, Any],
    ):
        for document in self.documents:
            if _matches(document, query):
                for key, value in update.get("$set", {}).items():
                    _set_nested(document, key, value)
                return SimpleNamespace(modified_count=1)
        return SimpleNamespace(modified_count=0)


class FakeDatabase:

from datetime import datetime, timedelta, timezone

import pytest

from backend.app.models.enums import RequestState

# Development/testing authentication tokens configured by the application.
IO_AUTH = {"Authorization": "Bearer dev-io-token"}
OFFICER_AUTH = {"Authorization": "Bearer dev-officer-token"}


def _make_request_document(
    *,
    request_id: str,
    status: RequestState,
    created_at: datetime,
) -> dict:
    """Create a valid persisted request document for endpoint tests."""
    document = LocationRequestDocument.create_new(
        request_id=request_id,
        case_id=f"CASE-{request_id}",
        target_phone_number="+919876543210",
        operator_code="JIO",
        submitting_officer_id="OFFICER-DEV-001",
    ).model_dump(mode="json")

    document["status"] = status.value
    document["created_at"] = created_at.isoformat()
    document["updated_at"] = created_at.isoformat()

    return document


@pytest.mark.asyncio
async def test_io_can_retrieve_pending_requests(
    async_client,
    fake_database,
):
    """An authenticated IO can retrieve requests awaiting review."""
    now = datetime.now(timezone.utc)

    pending_request = _make_request_document(
        request_id="REQ-PENDING-001",
        status=RequestState.PENDING_IO_REVIEW,
        created_at=now,
    )

    await fake_database.location_requests.insert_one(pending_request)

    response = await async_client.get(
        "/api/v1/requests/io/pending",
        headers=IO_AUTH,
    )

    assert response.status_code == 200

    payload = response.json()

    assert payload["success"] is True
    assert isinstance(payload["data"], list)
    assert len(payload["data"]) == 1
    assert payload["data"][0]["request_id"] == "REQ-PENDING-001"
    assert payload["data"][0]["status"] == "PENDING_IO_REVIEW"


@pytest.mark.asyncio
async def test_pending_queue_excludes_requests_in_other_states(
    async_client,
    fake_database,
):
    """Only PENDING_IO_REVIEW requests should appear in the IO queue."""
    now = datetime.now(timezone.utc)

    pending_request = _make_request_document(
        request_id="REQ-PENDING-002",
        status=RequestState.PENDING_IO_REVIEW,
        created_at=now,
    )

    created_request = _make_request_document(
        request_id="REQ-CREATED-001",
        status=RequestState.CREATED,
        created_at=now - timedelta(minutes=1),
    )

    executing_request = _make_request_document(
        request_id="REQ-EXECUTING-001",
        status=RequestState.EXECUTING,
        created_at=now - timedelta(minutes=2),
    )

    await fake_database.location_requests.insert_one(pending_request)
    await fake_database.location_requests.insert_one(created_request)
    await fake_database.location_requests.insert_one(executing_request)

    response = await async_client.get(
        "/api/v1/requests/io/pending",
        headers=IO_AUTH,
    )

    assert response.status_code == 200

    payload = response.json()

    assert payload["success"] is True

    returned_ids = [
        item["request_id"]
        for item in payload["data"]
    ]

    assert returned_ids == ["REQ-PENDING-002"]

    async_client,
    fake_database,

        request_id="REQ-NEWER-001",
        status=RequestState.PENDING_IO_REVIEW,
        created_at=now - timedelta(minutes=5),

        request_id="REQ-NEWEST-001",
        status=RequestState.PENDING_IO_REVIEW,
        created_at=now,
    )

    # Insert them in a different order to verify database sorting.
    await fake_database.location_requests.insert_one(older_request)
    await fake_database.location_requests.insert_one(newest_request)
    await fake_database.location_requests.insert_one(newer_request)

    response = await async_client.get(
        "/api/v1/requests/io/pending",
        headers=IO_AUTH,
    )

    assert response.status_code == 200

    returned_ids = [
        item["request_id"]
        for item in response.json()["data"]
    ]

    assert returned_ids == [
        "REQ-NEWEST-001",
        "REQ-NEWER-001",
        "REQ-OLDER-001",
    ]

    return events

"""Async PostgreSQL persistence adapter.

The adapter intentionally preserves the existing collection-style call surface used by
routes while executing all persistence through SQLAlchemy 2.x and psycopg 3.
Payloads retain the existing serialized Pydantic document shape; indexed relational
columns enforce identifiers and support common lookups. Nested fields remain JSONB
because they are existing structured lifecycle objects.
"""
from __future__ import annotations

from datetime import datetime
from typing import Any, Dict, Optional
from sqlalchemy import (
    MetaData, Table, Column, String, DateTime, Boolean, select, insert, update,
    text, Index,
)
from sqlalchemy.dialects.postgresql import JSONB
from sqlalchemy.ext.asyncio import AsyncEngine, AsyncSession, async_sessionmaker, create_async_engine
from sqlalchemy.exc import SQLAlchemyError

from backend.app.core.config import Settings, get_settings
from backend.app.core.exceptions import DatabaseConnectionError
from backend.app.core.logging import get_logger

logger = get_logger(__name__)
metadata = MetaData()

location_requests_table = Table(
    "location_requests", metadata,
    Column("request_id", String(80), primary_key=True),
    Column("case_id", String(255), nullable=False),
    Column("status", String(64), nullable=False, index=True),
    Column("operator_code", String(32), nullable=False, index=True),
    Column("submitting_officer_id", String(255), nullable=False, index=True),
    Column("share_token_hash", String(64), unique=True, nullable=True),
    Column("created_at", DateTime(timezone=True), nullable=False, index=True),
    Column("updated_at", DateTime(timezone=True), nullable=False),
    Column("payload", JSONB, nullable=False),
)
Index("ix_location_requests_status_created", location_requests_table.c.status, location_requests_table.c.created_at)

operators_table = Table(
    "operators", metadata,
    Column("operator_code", String(32), primary_key=True),

    Column("payload", JSONB, nullable=False),

audit_logs_table = Table(
    "audit_logs", metadata,
    Column("audit_id", String(80), primary_key=True),
    Column("request_id", String(80), nullable=True),
    Column("timestamp", DateTime(timezone=True), nullable=False, index=True),
    Column("actor_id", String(255), nullable=False, index=True),
    Column("event_type", String(80), nullable=False, index=True),

TABLES = {
    "location_requests": location_requests_table,

    "audit_logs": audit_logs_table,

def _nested(data: dict, path: str):
    cur = data
    for part in path.split("."):
        if not isinstance(cur, dict) or part not in cur:
            return None
        cur = cur[part]
    return cur

def _matches(data: dict, query: dict) -> bool:
    for key, expected in query.items():
        if _nested(data, key) != expected:
            return False
    return True

def _flatten_payload(payload: dict, table: Table) -> dict:
    p = dict(payload)
    if table is location_requests_table:
        token = p.get("share_token") or {}
        return {
            "request_id": p["request_id"], "case_id": p["case_id"],
            "status": getattr(p.get("status"), "value", p.get("status", "CREATED")),
            "operator_code": p["operator_code"],
            "submitting_officer_id": p["submitting_officer_id"],
            "share_token_hash": token.get("token_hash"),
            "created_at": _dt(p.get("created_at")), "updated_at": _dt(p.get("updated_at")),
            "payload": p,
        }
    if table is operators_table:
        return {"operator_code": p["operator_code"], "is_active": bool(p.get("is_active", True)), "payload": p}
    return {
        "audit_id": p["audit_id"], "request_id": p.get("request_id"),
        "timestamp": _dt(p.get("timestamp")), "actor_id": p["actor_id"],
        "event_type": getattr(p.get("event_type"), "value", p.get("event_type")),
        "payload": p,
    }

def _dt(value):
    if isinstance(value, datetime):
        return value
    if isinstance(value, str):
        return datetime.fromisoformat(value.replace("Z", "+00:00"))
    return value

def _document(row):
    if row is None:
        return None
    payload = dict(row["payload"])
    return payload

class AsyncCursor:
    def __init__(self, collection, query):
        self.collection, self.query, self.order = collection, query, None
    def sort(self, key, direction):
        self.order = (key, direction)
        return self
    def __aiter__(self):
        async def iterator():
            for item in await self.collection._find(self.query, self.order):
                yield item
        return iterator()

class PostgresCollection:
    def __init__(self, manager, name):
        self.manager, self.name, self.table = manager, name, TABLES[name]
    async def insert_one(self, document):
        async with self.manager.sessions() as session:
            async with session.begin():
                await session.execute(insert(self.table).values(**_flatten_payload(document, self.table)))
        return type("InsertResult", (), {"inserted_id": document.get("request_id", document.get("audit_id", document.get("operator_code")))})()
    async def find_one(self, query, projection=None):
        rows = await self._find(query, None, limit=1)
        return rows[0] if rows else None
    def find(self, query):
        return AsyncCursor(self, query)
    def _conditions(self, query):
        conditions = []
        for key, value in query.items():
            mapped = {"share_token.token_hash": "share_token_hash"}.get(key, key)
            if mapped in self.table.c:
                conditions.append(self.table.c[mapped] == value)
        return conditions

    async def _find(self, query, order, limit=None):
        async with self.manager.sessions() as session:
            stmt = select(self.table).where(*self._conditions(query))
            result = await session.execute(stmt)
            docs = []
            for row in result.mappings():
                doc = _document(row)
                if _matches(doc, query):
                    docs.append(doc)
            if order:
                key, direction = order
                docs.sort(key=lambda d: _nested(d, key) or "", reverse=direction < 0)
            return docs[:limit] if limit else docs
    async def update_one(self, query, update_doc):
        # Serialize candidate selection/update in one transaction. Row-level locking
        # prevents concurrent lifecycle transitions from overwriting one another.
        async with self.manager.sessions() as session:
            async with session.begin():
                stmt = select(self.table).where(*self._conditions(query)).with_for_update()
                result = await session.execute(stmt)
                for row in result.mappings():
                    doc = _document(row)
                    if not _matches(doc, query):
                        continue
                    for path, value in update_doc.get("$set", {}).items():
                        cur = doc
                        parts = path.split(".")
                        for part in parts[:-1]:
                            if not isinstance(cur.get(part), dict):
                                cur[part] = {}
                            cur = cur[part]
                        cur[parts[-1]] = value.isoformat() if isinstance(value, datetime) else value
                    vals = _flatten_payload(doc, self.table)
                    key_col = list(self.table.primary_key.columns)[0]
                    key = row[key_col.name]
                    await session.execute(update(self.table).where(key_col == key).values(**vals))
                    return type("UpdateResult", (), {"modified_count": 1})()
        return type("UpdateResult", (), {"modified_count": 0})()

class DatabaseManager:
    def __init__(self):
        self.engine: Optional[AsyncEngine] = None
        self.sessions = None
        self.is_connected = False
        self._collections = {name: PostgresCollection(self, name) for name in TABLES}
    async def connect(self, settings: Optional[Settings] = None) -> bool:
        cfg = settings or get_settings()
        try:
            self.engine = create_async_engine(
                cfg.database_url,
                pool_size=cfg.database_pool_size,
                max_overflow=cfg.database_max_overflow,
                pool_timeout=cfg.database_pool_timeout_seconds,
                pool_pre_ping=True,
            )
            self.sessions = async_sessionmaker(self.engine, expire_on_commit=False, class_=AsyncSession)
            async with self.engine.connect() as conn:
                await conn.execute(text("SELECT 1"))
            self.is_connected = True
            logger.info("Connected to PostgreSQL.")
            return True
        except Exception as exc:
            self.is_connected = False
            logger.warning("PostgreSQL unavailable (%s); running in degraded/offline mode.", type(exc).__name__)
            return False
    async def close(self):
        if self.engine is not None:
            await self.engine.dispose()
        self.engine = None
        self.sessions = None
        self.is_connected = False
    async def check_health(self) -> Dict[str, Any]:
        if self.engine is None or self.sessions is None:
            return {"status": "disconnected", "message": "Database engine has not been initialized."}
        try:
            async with self.engine.connect() as conn:
                await conn.execute(text("SELECT 1"))
            self.is_connected = True
            return {"status": "connected", "database": get_settings().database_name}
        except Exception as exc:
            self.is_connected = False
            return {"status": "disconnected", "message": f"Connection check failed: {type(exc).__name__}"}
    def get_database(self):
        if not self.is_connected or self.sessions is None:
            raise DatabaseConnectionError("PostgreSQL database is not initialized or unavailable.")
        return self
    @property
    def location_requests(self):
        db = self.get_database()
        return db._collections["location_requests"] if db is self else db["location_requests"]
    @property
    def operators(self):
        db = self.get_database()
        return db._collections["operators"] if db is self else db["operators"]
    @property
    def audit_logs(self):
        db = self.get_database()
        return db._collections["audit_logs"] if db is self else db["audit_logs"]

db_manager = DatabaseManager()
async def get_database():
    return db_manager.get_database()

"""FastAPI Application Entrypoint for IFSO Location Request Management System.

Configures lifespan lifecycle events, logging, database connections, CORS,
exception handlers, and API routing.
"""

from contextlib import asynccontextmanager
from typing import AsyncGenerator
from fastapi import FastAPI, status
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import JSONResponse

from backend.app.core.config import get_settings
from backend.app.core.database import db_manager
from backend.app.core.exceptions import register_exception_handlers
from backend.app.core.logging import get_logger, setup_logging
from backend.app.routes.api import api_v1_router

logger = get_logger("app.main")


@asynccontextmanager
async def lifespan(app: FastAPI) -> AsyncGenerator[None, None]:
    """Application lifespan manager for startup and shutdown routines."""
    settings = get_settings()

    # Configure structured redacting logger
    setup_logging(log_level=settings.log_level)
    logger.info(
        f"Starting {settings.app_name} [v{settings.app_version}] "
        f"in '{settings.app_env}' environment (debug={settings.app_debug})."
    )

    # Attempt database connection
    await db_manager.connect(settings)

    yield

    # Teardown database connection pool
    logger.info("Initiating graceful shutdown...")
    await db_manager.close()
    logger.info("Application shutdown complete.")


def create_application() -> FastAPI:
    """Application factory for FastAPI instance."""
    settings = get_settings()

    app = FastAPI(
        title=settings.app_name,
        description=(
            "REST API for the IFSO Location Request Management System. "
            "Automates authorized SMS location queries with tamper-evident audit trails."
        ),
        version=settings.app_version,
        docs_url="/docs",
        redoc_url="/redoc",
        openapi_url="/openapi.json",
        lifespan=lifespan,
        debug=settings.app_debug,
    )

    # CORS configuration for development and mobile clients
    app.add_middleware(
        CORSMiddleware,
        allow_origins=["*"],
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    # Register centralized exception handlers
    register_exception_handlers(app)

    # Root health check endpoint
    @app.get(
        "/",
        tags=["System & Health"],
        status_code=status.HTTP_200_OK,
        summary="Root health and metadata endpoint",
    )
    async def root_status() -> JSONResponse:
        db_health = await db_manager.check_health()
        return JSONResponse(
            status_code=status.HTTP_200_OK,
            content={
                "status": "healthy" if db_health.get("status") == "connected" else "degraded",
                "app": settings.app_name,
                "version": settings.app_version,
                "environment": settings.app_env,
                "database_status": db_health.get("status"),
                "docs_url": "/docs",
                "api_v1": settings.api_v1_prefix,
            },
        )

    # Register API routes under v1 prefix
    app.include_router(api_v1_router, prefix=settings.api_v1_prefix)

    return app


# Application singleton for ASGI servers (uvicorn backend.app.main:app)
app = create_application()
